$$

$$

$$

model Lorenz

$$
\begin{array}{l} \text { parameter   Real   sigma } = 1 0; \\ \text { parameter   Real   rho } = 2 8; \\ \text { parameter   Real   beta } = 8 / 3; \\ \text { Real } x (\text { start } = 1. 0); \\ \text { Real } y (\text { start } = 0. 0); \\ \text { Real } z (\text { start } = 0. 0); \\ \end{array}
$$

equation

$$
\begin{array}{l} \operatorname{der} (x) = \text { sigma } * (y - x); \\ \operatorname{der} (y) = x * (\text { rho } - z) - y; \\ \operatorname{der} (z) = x ^ {*} y - \text { beta } ^ {*} z; \\ \end{array}
