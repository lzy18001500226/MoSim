```julia
# 计算有限值的均方根误差
rmse(values) = begin
    finite = finite_values(values)
    isempty(finite) ? NaN : sqrt(mean_value(finite .^ 2))
end

# 三维位置误差分解
ex = x .- x_ref
ey = y .- y_ref
ez = z .- z_ref
ep = sqrt.(ex .^ 2 .+ ey .^ 2 .+ ez .^ 2)

# 计算位置RMSE
position_rmse = rmse(ep)
```

**控制能量积分**：

```julia
# 梯形积分法
function trapezoid_integral(time, values)
    length(time) < 2 && return NaN
    length(values) < 2 && return NaN
    
    total = 0.0
    for index in 2:length(time)
        dt = time[index] - time[index - 1]
        isfinite(dt) && dt > 0 || continue
        total += 0.5 * dt * (values[index] + values[index - 1])
    end
    
    return total
end

# 控制输入二范数平方：||u(t)||² = u₁² + u₂² + u₃² + u₄²
motor_cols = [name for name in MOTOR_COLUMNS if haskey(columns, name)]
control_norm_sq = zeros(length(time))
if !isempty(motor_cols)
    for name in motor_cols
        control_norm_sq .+= columns[name] .^ 2
    end
end

# 控制能量 = ∫₀ᵀ ||u(t)||² dt
metrics["control_energy"] = trapezoid_integral(time, control_norm_sq)
```

**说明**：
- 全部使用Base Julia实现，无第三方依赖
- 可直接在MWORKS Syslab环境中运行
- RMSE公式：√(Σ(x-x_ref)²/N)
- 控制能量公式：∫₀ᵀ(u₁²+u₂²+u₃²+u₄²)dt

```julia
function write_climb_rmse_bar(path::String, climb_rows)
    # 画布尺寸与边距
    width, height = 980, 620
    left, right, top, bottom = 140.0, 40.0, 55.0, 135.0
    
    # 提取数据
    labels = [row["controller_id"] for row in climb_rows]
    values = [Float64(row["metrics"]["position_rmse_m"]) for row in climb_rows]
    
    # 计算绘图区域
    plot_width = width - left - right
    plot_height = height - top - bottom
    y_max = maximum(filter(isfinite, values)) * 1.1
    bar_width = plot_width / length(labels) * 0.7
    
    # 直接写入SVG
    open(path, "w") do io
        println(io, """<?xml version="1.0" encoding="UTF-8"?>""")
        println(io, """<svg xmlns="http://www.w3.org/2000/svg" width="$width" height="$height">""")
        
        # Y轴标签（18pt，Times New Roman）
        println(io, """<text x="50" y="$(height/2)" text-anchor="middle" """
                  * """transform="rotate(-90 50 $(height/2))" """
                  * """font-family="Times New Roman" font-size="18">"""
                  * """Position RMSE (m)</text>""")
        
        # 绘制柱状图
        for (index, (label, value)) in enumerate(zip(labels, values))
            center_x = left + (index - 0.5) * plot_width / length(labels)
            bar_height = isfinite(value) ? value / y_max * plot_height : 0.0
            y = height - bottom - bar_height
            color = COLORS[mod1(index, length(COLORS))]
            
            # 柱体
            println(io, """<rect x="$(center_x - bar_width/2)" y="$y" """
                      * """width="$bar_width" height="$bar_height" fill="$color"/>""")
            
            # X轴标签（18pt，旋转-45°）
            println(io, """<text x="$center_x" y="$(height - bottom + 20)" """
                      * """text-anchor="end" """
                      * """transform="rotate(-45 $center_x $(height - bottom + 20))" """
                      * """font-family="Times New Roman" font-size="18">"""
                      * """$(svg_escape(label))</text>""")
            
            # 柱顶数值（16pt）
            println(io, """<text x="$center_x" y="$(y - 5)" text-anchor="middle" """
                      * """font-family="Times New Roman" font-size="16">"""
                      * """$(round(value, digits=3))</text>""")
        end
        
        println(io, "</svg>")
    end
end
```

**字体标准**：
- Y轴标签：18pt Times New Roman
- Y轴刻度：16pt Times New Roman  
- X轴标签：18pt Times New Roman（旋转-45°）
- 柱顶数值：16pt Times New Roman

**说明**：
- 不依赖Plots.jl等绘图库，直接生成SVG XML
- 矢量图格式，无损缩放，适合学术出版

```cpp
class MoSimAdapter {

        // 订阅里程计与参考轨迹

        // 提取4维参考向量
        reference_[0] = msg->pose.position.x;
        reference_[1] = msg->pose.position.y;
        reference_[2] = msg->pose.position.z;
```
