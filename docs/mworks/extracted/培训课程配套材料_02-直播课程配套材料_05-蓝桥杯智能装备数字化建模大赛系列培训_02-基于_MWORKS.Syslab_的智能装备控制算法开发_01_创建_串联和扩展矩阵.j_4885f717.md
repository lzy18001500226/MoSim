## 扩展矩阵

#扩展矩阵到mxn
function extend_mtx(A, m, n)
    B = A
    sz = size(B)
    A = zeros(eltype(A), m, n)
    A[1:sz[1], 1:sz[2]] = B
    return A
end

A = [10 20 30; 60 70 80]
A = extend_mtx(A, 3, 4)
A[3, 4] = 1
A
