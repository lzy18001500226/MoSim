using PyCall

#case 1
math = pyimport("math")

v = math.sin(pi/2)
println("v = $v")

#case 2
using TyPlot
@pyimport numpy as np
x = np.linspace(0, 2π, 1000)
y = np.sin(x)
plot(x,y)
