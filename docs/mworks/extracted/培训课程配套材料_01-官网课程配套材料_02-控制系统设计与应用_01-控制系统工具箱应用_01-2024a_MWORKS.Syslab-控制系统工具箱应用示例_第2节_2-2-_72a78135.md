Gd = c2d(G, 0.1)

Fig = gca().lines

##【示例12】将离散时间系统进行重采样，重采样时间为 ts = 0.02s
H = tf([1, -1], [1, 1, 0.3], 0.1)
# 对原系统进行重采样
Hc = d2d(H, 0.02)
# 阶跃响应对比
step(H, 1, "-b", linewidth=1.5)
step(Hc, 1, "-r", ishold=true, linewidth=1)
grid("on")
ylim([-2, 3.5])
Fig = gca().lines
legend([Fig[1], Fig[3]], ["原系统-采样时间0.1", "重采样系统-采样时间0.02"])
# 频率响应对比

w = (10^-1, 10^3)
bode(H, w)
bode(Hc, w, ishold=true)
bodegrid(true)
legend(["原系统-采样时间0.1", "Nyquist频率", "重采样系统-采样时间0.02"])
