B = [1 2 2;
    1 5 3;
    1 8 4;
    2 1 6;
    2 9 3;
    3 3 4;
    3 7 5;
    4 4 8;
    4 6 6;
    5 1 8;
    5 5 1;
    5 9 6;
    6 4 7;
    6 6 5;
    7 3 7;
    7 7 6;
    8 1 4;
    8 9 8;
    9 2 3;
    9 5 4;
    9 8 2];
using TyOptimization
using TyBase
sudpuzzle = optimproblem()
#创建一个优化变量 x，它是一个大小为 9×9×9 的二元变量。
x = optimvar("x", 9, 9, 9, Type="integer", Lowerbnd=0, Upperbnd=1)
#创建一个具有任意目标函数的优化问题
mul = ones(1, 1, 9);
mul = cumsum(mul, dims=3);
sudpuzzle.Objective = @fcn2expr sum(sum(sum(x, dims=1), dims=2) .* mul) x;
#sudpuzzle.Objective =@fcn2expr x
#将约束表示为：在每个坐标方向上 x 的总和为 1,这里需要注意，单个约束，左右两侧的维度需要一致
cons1 = sum(x, dims=1) == ones(1, 9, 9);
cons2 = sum(x, dims=2) == ones(9, 1, 9);
cons3 = sum(x, dims=3) == ones(9, 9, 1);


#还要创建粗线网格总和等于 1 的约束。
#按照案例
# ma_test=(3,3,3)
# majorg = optimconstr(==, ma_test);
majorg = optimconstr(==, 3, 3);
#约束赋值不支持使用向量式语句
for u = 1:3
    for v = 1:3
        arr = x[3*(u-1)+1:3*(u-1)+3, 3*(v-1)+1:3*(v-1)+3, :]
        majorg[u, v] = sum(sum(arr, dims=1), dims=2) == ones(1, 1, 9)
    end
end
show(majorg)
sudpuzzle.Constraints = (cons1=cons1, cons2=cons2, cons3=cons3, majorg=majorg);
#添加约束方式与MATLAB不同，这里需要一次性添加所有约束
#在包含提示的格点处设置下界 1 来包括初始提示值。此设置将对应项的值固定为 1，从而将每个提示数处的解设置为提示数本身。
for u = 1:size(B, 1)
    x.LowerBound[B[u, 1], B[u, 2], B[u, 3]] = 1
end

#检查问题
show(sudpuzzle)

#求解数独谜题
sudsoln, = Solve(sudpuzzle)

#显示求解结果
y = ones(size(sudsoln.x));
for k = 2:9
    y[:, :, k] .= k
end
using TyPlot
include("drawsudoku.jl")
S = sudsoln.x .* y;
S = sum(S, dims=3);
S = squeeze(S) #删除长度为 1 的维度
drawSudoku(Int.(S))
