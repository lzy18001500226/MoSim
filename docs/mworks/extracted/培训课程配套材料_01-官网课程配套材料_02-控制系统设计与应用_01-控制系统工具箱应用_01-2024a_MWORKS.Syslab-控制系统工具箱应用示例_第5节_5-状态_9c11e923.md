##【示例38】当控制对象传递函数具有一个积分器时，试设计一个I型伺服系统。设控制对象传递函数为：
#Y(s)/U(s)=1/s(s+1)(s+2)，希望的闭环极点为：𝑠=−2±𝑗∙2√3 ，和 s=−10。设计系统状态反馈增益𝑲，并计算系统单位阶跃响应
# 定义被控对象
s = tf("s")
G = 1 / (s * (s + 1) * (s + 2))
# 将被控对象转化为状态空间模型
Gss = ss(G)
# 定义闭环系统期望极点
idealpoles = [-2 + im * 2 * sqrt(3), -2 - im * 2 * sqrt(3), -10]
# 使用place函数求解状态反馈增益
K = place(Gss.A, Gss.B, idealpoles)[1]
K = real.(K)
# 加入状态反馈的闭环系统计算
AA = Gss.A - Gss.B * K
BB = Gss.B * K[1]
CC = Gss.C
DD = Gss.D
# 被控对象单位闭环阶跃响应与状态反馈闭环阶跃计算对比
step(feedback(G), 15, linewidth=2)
step(ss(AA, BB, CC, DD), 15, linewidth=2, ishold=true)
grid("on")
Fig = gca().lines
legend([Fig[1], Fig[3]], ["原始系统单位闭环阶跃响应", "状态反馈系统阶跃响应"])

##【示例40】一维倒立摆系统基于极点配置法的状态反馈控制律设计
# 倒立摆系统参数
M = 2     # 小车质量2Kg
m = 0.1   # 摆球质量0.1Kg
l = 0.5   # 摆杆长度0.5m
g = 9.8   # 重力加速度9.8m/s^2
# 倒立摆系统状态空间矩阵定义
A = [0 1 0 0
    (M+m)*g/(M*l) 0 0 0
    0 0 0 1
    -m*g/M 0 0 0]
B = [0 -1 / (M * l) 0 1 / M]'
C = [0 0 1 0]
D = 0
# 倒立摆系统模型
G = ss(A, B, C, D)

# 构造误差状态方程，并判定系统能控性 
Ahat = [A zeros(size(A, 1), 1); -C 0]
Bhat = [B; 0]
r = rank([Ahat Bhat])

# 计算理想主导极点位置并分配系统所有极点位置 
idealPoles = roots([1, 2 * 0.7 * 1, 1])
AllPoles = [idealPoles[1], idealPoles[2], -10, -15, -20]
# 极点配置，计算状态增益K 
Khat = place(Ahat, Bhat, AllPoles)[1]
# 状态增益值分解 
K = reshape(real(Khat[1:4]), 1, 4)
k1 = -real(Khat[end])

# 闭环系统状态方程 
AA = [A-B*K B*k1; -C 0]
BB = [0 0 0 0 1]'
CC = [C 0]
DD = 0
Gclose = ss(AA, BB, CC, DD)

# 计算并绘制阶跃响应 
step(Gclose, 10, linewidth=2)
grid("on")

# 计算并绘制角度变化
y, t, x = step(Gclose, 10, fig=false)
plot(t, rad2deg.(x[1, :]), "-r", linewidth=2)
grid("on")
title(raw"摆杆夹角$\theta$")
xlabel("时间(秒)")
ylabel("度(deg)")
