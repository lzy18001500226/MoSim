        messageBox("1,创建实体失败");

{
        messageBox("2,创建实体成功");
        test = false;
}
else
{
        messageBox("2,创建实体失败");
}

示例2：
''''''动态创建--------------------------
void 动态创建()
{
int i=1                                                                                                                         ''''定义整型变量i，将1赋值给i
string str;                                                                                                                     ''''定义字符串变量str
while(i<=6){                                                                                                            ''''当整型变量i不大于6时，执行while循环
        string s_i = itoa(i);                                                                                   ''''整型变量i转化为字符串变量s_i
        str ="烟雾发射器" + s_i;                                                                             ''''字符串变量str = "烟雾发射器" + s_i

        createEntity(str,"烟雾发射器")                                                                       ''''当实体建立的时候调用----------------------
        setValueString(str,"s_CurrentModuleName",str)                                   ''''设置字符串变量数值
        setValueInt(str,"carrierNumber",i)                                                              ''''设置整型变量数值
        showField("/scene/modules/main/"+str+"/model/烟雾系统.vis",1)       ''''设置整型变量数值
        i++;                                                                                                                    ''''整型变量+1
}
